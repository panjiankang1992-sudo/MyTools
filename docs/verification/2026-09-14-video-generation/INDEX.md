```text
<窗口>/

  host-before.json     # 启动前的内存与显存门槛实测
  host-after.json      # 结束后的资源状态
  schema.sha256        # 固定运行时 /object_info 的哈希
  cases.jsonl          # 每个用例：参数、耗时、产物哈希、峰值资源、客观指标
  inputs/manifest.json # 输入素材的字节数与 SHA-256
  outputs/<run>/       # output.mp4、workflow.json、ffprobe、review-metrics、关键帧、接触表、inputs.json
  metrics/<run>/       # metrics.jsonl（逐次采样的显存/内存/PSI）、host-before/after
  reviews.json         # 人工评分模板（当前全部 pending）
  report.md            # 自动汇总
```
